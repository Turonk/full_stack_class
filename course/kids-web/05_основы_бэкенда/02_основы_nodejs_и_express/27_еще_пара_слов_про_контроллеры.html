Ещё пара слов про контроллеры

Напомним, **контроллер** — это функция, предназначенная для выполнения определённых действий в качестве реакции на полученный запрос по какому-либо маршруту.

// routes/games.js
const gamesRouter = require("express").Router(); // Создали роутер
const { readData, writeData } = require("../utils/data"); // Чтение и запись данных в json файл

// Получим игры из JSON файла и отправим в ответ на запрос
const getAllGames = async (req, res) => {
  const games = await readData("./data/games.json");
  if (!games) {
    res.status(400);
    res.send({
      status: "error",
      message: "Нет игр в базе данных. Добавьте игру."
    });
    return;
  }
  req.games = games;
  res.send(req.games);
};

const deleteGame = async (req, res) => {
	// Получим данные из файла
  const games = await readData("./data/games.json");
  if (!games) {
    res.status(400);
    res.send({
      status: "error",
      message: "Нет игр в базе данных. Добавьте игру.",
    });
    return;
  }
  req.games = games;
  
  // Прочитаем запрашиваемый id игры из запроса
	const id = Number(req.params.id);
	
	// Найдём игру, которую хотят удалить из общего массива с играми, по id
	req.game = req.games.find((item) => item.id === id);
	
	// Найдём индекс удаляемой игры в общем массиве игр
	const index = req.games.findIndex((item) => item.id === req.game.id);
	
	// Удалим из массива игр игру
	req.games.splice(index, 1);
	
	// Запишем обновленный массив игр в JSON файл
	await writeData("./data/games.json", req.games);
	
	// Вернём ответ о проделанной операции с данными о играх
	res.send({
    games: req.games,
    updated: req.game
  });
}

gamesRouter.get("/games", getAllGames)
gamesRouter.delete("/games/:id", deleteGame);

module.exports = gamesRouter;

В этом случае функции `getAllGames` и `deleteGame` — контроллеры, которые используются для обработки маршрутов 👇🏻

gamesRouter.get("/games", getAllGames)
gamesRouter.delete("/games/:id", deleteGame);

{quiz-task} background: | #f1f1ef content: | ❗ Но пока в данном коде есть проблема: когда в `routes/games.js` появится больше маршрутов, хранить все контроллеры для них в этом же файле будет очень неудобно, тем более если сами контроллеры будут сложные и объёмные. {/quiz-task}

Создай отдельную папку `controllers` с `games.js` , перенеси в него функции `getAllGames` и `deleteGame`. Дальше будешь экспортировать их оттуда, чтобы можно было использовать в файле с маршрутами `routes/games.js` 👇🏻
