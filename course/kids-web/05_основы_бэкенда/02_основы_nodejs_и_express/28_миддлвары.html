Что такое `middleware`

`middleware` — это функция, которая вызывается в промежуток между тем, как сервер получил запрос и отправил ответ на него. Иными словами, перед тем, как сработает контроллер.

`middleware`-функция в качестве аргументов принимает:

- Объект запроса `req` — из этого объекта можно получить доступ ко всем параметрам полученного запроса. - Объект ответа `res` — предназначен для отправки ответа сервером. - Функцию `next` — рассмотрим чуть позже, для чего она нужна.

В качестве примера лови маленькое серверное приложение на `express` 👇🏻

const express = require("express");
const app = express();

// Объявили middleware-функцию loggingMiddleware
function loggingMiddleware(req, res, next) {
  console.log("Внутри loggingMiddleware");
}

// Контроллер для маршрута /
function homeController(req, res) {
    console.log("Внутри homeController");
    res.send("Home Page");
}

// Контроллер для маршрута /users
function usersController(req, res) {
    console.log("Внутри usersController");
    res.send("Users Page");
}

// Обрабатываем запросы методом GET по маршруту "/"
app.get("/", homeController);

// Обрабатываем запросы методом GET по маршруту "/users"
app.get("/users", usersController);

app.listen(3000, () => console.log("Server Started"));

В коде выше мы обрабатывали запросы по маршрутам `/` и `/users`, а также объявили `middleware`-функцию `loggingMiddleware`. Сейчас при получении запроса по любому маршруту `loggingMiddleware` не будет вызываться, так как мы её не подключили.

Чтобы это сделать, используем метод `use` 👇🏻

const express = require("express");
const app = express();

// Объявили middleware-функцию loggingMiddleware
function loggingMiddleware(req, res, next) {
  console.log("Внутри loggingMiddleware");
}

// Контроллер для маршрута /
function homeController(req, res) {
    console.log("Внутри homeController");
    res.send("Home Page");
}

// Контроллер для маршрута /users
function usersController(req, res) {
    console.log("Внутри usersController");
    res.send("Users Page");
}

// Глобально подключили middleware-функцию loggingMiddleware
app.use(loggingMiddleware);

// Обрабатываем запросы методом GET по маршруту "/"
app.get("/", homeController)
// Обрабатываем запросы методом GET по маршруту "/users"
app.get("/users", usersController);

app.listen(3000, () => console.log("Server Started"));

Строкой `app.use(loggingMiddleware)` мы глобально подключили `loggingMiddleware`. Глобально — значит, что функция `loggingMiddleware` будет вызываться перед контроллерами по всем маршрутам в приложении, то есть в нашем случае перед контроллерами `homeController` и `usersController`

{video:https://code.s3.yandex.net/kids-web/vids/M4/8.mp4|controls:true|autoplay:0|pip:false|controlsList:nodownload}

На скринкасте показаны запуск приложения (код выше) и выполнение запроса из браузера по адресу `http://localhost:3000/`. В результате в консоли появляется строка `Внутри loggingMiddleware`, но никакой `Внутри homeController` не будет.

Так произошло, потому что внутри `loggingMiddleware` мы не вызываем функцию `next` — если добавить её вызов, то всё сработает как должно 👇🏻

{video:https://code.s3.yandex.net/kids-web/vids/M4/9.mp4|controls:true|autoplay:0|pip:false|controlsList:nodownload}

В нашем небольшом приложении обработка запросов на каждый маршрут уже состоит из двух этапов: `middleware`-функция `loggingMiddleware` и контроллер. Так вот функция `next` как раз и предназначена для того, чтобы передать управление от одного этапа к следующему. Функцию `next` не нужно путать с `return`: ключевое слово `return` полностью остановит дальнейший процесс обработки запроса, функция `next`, наоборот, передаёт эстафету следующим обработчикам.

Давай немного усложним наш пример: теперь у нас есть две `middleware`-функции: `nameMiddleware` и `greetingMiddleware` 👇🏻

const express = require("express");
const app = express();

// Объявили middleware-функцию nameMiddleware
function nameMiddleware(req, res, next) {
  const name = 'aski';
  console.log(`Меня зовут ${name}`)
  req.name = name
  next();
}

// Объявили middleware-функцию greetingMiddleware
function greetingMiddleware(req, res, next) {
  if (req.name === undefined) {
    console.log(`А как тебя зовут?`);
  } else {
    console.log(`Привет! ${req.name}`);
  }
  next();
}

// Объявили контроллер для маршрута /
function homeController(req, res) {
  res.send("Home Page");
}

// Объявили контроллер для маршрута /users
function usersController(req, res) {
  res.send("Users Page");
}

// Глобально подключили middleware-функцию nameMiddleware
app.use(nameMiddleware);
// Глобально подключили middleware-функцию greetingMiddleware
app.use(greetingMiddleware);
// Обрабатываем запросы методом GET по маршруту "/"
app.get("/", homeController);
// Обрабатываем запросы методом GET по маршруту "/users"
app.get("/users", usersController);
app.listen(3000, () => console.log("Server Started"));

Обрати внимание на строку `req.name = name;` в `nameMiddleware`. Тут мы присваиваем объекту запроса новое поле `name`, которое в `greetingMiddleware` непосредственно и используется.

{quiz-task} background: | #f1f1ef content: | ❗ В этом основной смысл `middleware`-функций: они получают доступ к объекту запроса `req` и ответа `res` и могут добавлять или модифицировать данные в них. Таким образом, объекты `req` и `res` проходят через все `middleware`-функции и доходят до функции контроллера. {/quiz-task}

Работа контроллера `homeController` и ответ браузеру на его запрос текстом `Home Page`

Есть ещё один момент, требующий твоего внимания: порядок подключения миддлваров функцией `use` напрямую влияет на порядок их выполнения при обработке запроса. Посмотри на скринкаст, и станет яснее 👇🏻

{video:https://code.s3.yandex.net/kids-web/vids/M4/999.mp4|controls:true|autoplay:0|pip:false|controlsList:nodownload}

На видео мы поменяли местами подключения двух `middleware`

Было 👇🏻

// Глобально подключили middleware-функцию nameMiddleware
app.use(nameMiddleware);
// Глобально подключили middleware-функцию greetingMiddleware
app.use(greetingMiddleware);

Стало 👇🏻

// Глобально подключили middleware-функцию greetingMiddleware
app.use(greetingMiddleware);
// Глобально подключили middleware-функцию nameMiddleware
app.use(nameMiddleware);

Функция `greetingMiddleware` теперь всегда будет выполняться перед `nameMiddleware`. То есть в момент срабатывания `greetingMiddleware` в объекте `req` ещё нет поля `name`, и поэтому в консоль выводится `А как тебя зовут?`

Стоит также упомянуть, что если у тебя больше одного миддлвара, то необязательно для подключения каждого использовать отдельный вызов `app.use()`. Все миддлвары можно передать в один-единственный `app.use()`. Вот так можно было бы подключить `greetingMiddleware` и `nameMiddleware` 👇🏻

app.use(greetingMiddleware, nameMiddleware);

Такой способ полностью идентичен предыдущему способу подключать миддлвары, когда каждый из них передаётся в отдельный `app.use()`. Порядок следования миддлваров в одном `app.use()` тоже влияет на порядок их выполнения.

Ок! Реализуем 👇🏻

const express = require("express");
const app = express();

// Объявили middleware-функцию nameMiddleware
function nameMiddleware(req, res, next) {
  const name = 'Aski';
  console.log(`Меня зовут ${name}`)
  req.name = name
  next();
}

// Объявили middleware-функцию greetingMiddleware
function greetingMiddleware(req, res, next) {
  if (req.name === undefined) {
    console.log(`А как тебя зовут?`);
  } else {
    console.log(`Привет! ${req.name}`);
  }
  next();
}

// Объявили middleware-функцию для проверки авторизации
function authMiddleware(req, res, next) {
  console.log(`Пользователь ${req.name} авторизован!`);
  req.isAuth = true;
  next()
}

// Объявили контроллер для маршрута /
function homeController(req, res) {
  if (req.isAuth) {
    res.send(`${req.name} home page`);
  } else {
    res.send("Home Page");
  }
}

// Объявили контроллер для маршрута /users
function usersController(req, res) {
  res.send("Users Page");
}

// Глобально подключили middleware-функцию greetingMiddleware
app.use(greetingMiddleware);
// Глобально подключили middleware-функцию nameMiddleware
app.use(nameMiddleware);

// Обрабатываем запросы методом GET по маршруту "/"
app.get("/", authMiddleware, homeController);
// Обрабатываем запросы методом GET по маршруту "/users"
app.get("/users", usersController);
app.listen(3000, () => console.log("Server Started"));

Мы создали третью `middleware`-функцию `authMiddleware` и подключили её так 👇🏻

app.get("/", authMiddleware, homeController);

Теперь `authMiddleware` будет срабатывать только при запросе по маршруту `/`. На видео можешь глянуть, как это работает 👇🏻

Используем `middleware` в проекте

Прежде чем использовать миддлвары, посмотри на фронтенд проекта: в нём есть форма для добавления новой игры, после нажатия на кнопку `Добавить` фронтенд делает запрос методом `POST` по маршруту `/games` 👇🏻

{video:https://code.s3.yandex.net/kids-web/vids/M4/22.mp4|controls:true|autoplay:0|pip:false|controlsList:nodownload}

Сейчас `express`-приложение не умеет обрабатывать такие запросы и отвечает ошибкой с номером `404`. Чтобы исправить это, создай новый маршрут в файле `routes/games.js` для запросов методом `POST` по адресу `/games` 👇🏻

gamesRouter.post("/games", addGameController)

Для обработки запроса будем использовать контроллер `addGameController` 👇🏻

const addGameController = async (req, res) => {
    // Читаем список игр из файла
    const games = await readData("./data/games.json");
    if (!games) {
        res.status(400);
        res.send({
        status: "error",
        message: "Нет игр в базе данных. Добавьте игру.",
        });
        return;
    }
    req.games = games;
    // Проверяем, есть ли уже в списке игра с таким же названием
    req.isNew = !Boolean(req.games.find(item => item.title === req.body.title));
    // Если игра, которую хотим добавить, новая (её не было в списке)
    if (req.isNew) {
        // Добавляем объект с данными о новой игре
        const inArray = req.games.map(item => Number(item.id));
        let maximalId;
        if (inArray.length > 0) {
          maximalId = Math.max(...inArray);
        } else {
          maximalId = 0;
        }
        req.updatedObject = {
          id: maximalId + 1,
          title: req.body.title,
          image: req.body.image,
          link: req.body.link,
          description: req.body.description
        };
        // Добавляем данные о новой игре в список с другими играми
        req.games = [...req.games, req.updatedObject];
    } else {
        res.status(400);
        res.send({ status: "error", message: "Игра с таким именем уже есть." });
        return
    }
    // Записываем обновлённый список игр в файл
    await writeData("./data/games.json", req.games);
    // В качестве ответа отправляем объект с двумя полями
    res.send({
        games: req.games, // Обновлённый список со всеми играми
        updated: req.updatedObject // Новая добавленная игра
    });
}

Библиотека `body-parser`

Дело в том, что данные, которые были отправлены из фронтенда в виде `json`-объекта, нельзя просто так взять и прочитать из объекта `req.body`, сначала требуется их специальным образом подготовить, чтобы с ними можно было снова работать как c `json`-объектами. До того как мы начали изучать `express`, мы уже писали функцию `parseBody` с использованием конструктора `Promise`, которая выполняет эту задачу.

Сейчас же мы наконец подходим к `middleware`: для того чтобы из `req.body` можно было прочитать данные, необходимо воспользоваться миддлварами из библиотеки `body-parser`

Как мы это сделали, смотри в скринкасте 👇🏻

{video:https://code.s3.yandex.net/kids-web/vids/M4/55.mp4|controls:true|autoplay:0|pip:false|controlsList:nodownload}

Сначала установили саму библиотеку командой 👇🏻

npm install body-parser

Далее добавили её использование в файл `app.js` 👇🏻

// app.js 
const express = require('express');
const path = require('path');
const bodyParser = require('body-parser');
const mainRoute = require('./routes/main');
const gamesRouter = require('./routes/games'); 

const PORT = 3000;
const app = express();

app.use(
    bodyParser.json(),
    express.static(path.join(__dirname, 'public')),
    mainRoute,
    gamesRouter
); 

app.listen(PORT, () => {
    console.log(`Server is running at PORT http://localhost:${PORT}`);
})

Теперь `middleware`-функция `bodyParser.json` подключена глобально и будет вызываться для всех маршрутов приложения.

Выносим общий код в `middleware`

Обрати внимание на код всех трёх контроллеров: `deleteGame`, `addGameController` и `getAllGames`. В каждом из них есть один и тот же блок кода, который отвечает за чтение из файла `data/games.json` 👇🏻

const games = await readData("./data/games.json");
if (!games) {

    res.send({
    status: "error",

}

Его можно вынести в отдельную функцию и подключить как `middleware` в те маршруты, контроллеры которых используют `req.games`

Выполни следующие шаги:

1. Создай папку `middlewares` 2. В `middlewares` добавь файл `games.js`, в который положи функцию с общей частью кода, пусть эта функция будет называться `getAllGames` 3. Так как `getAllGames` — это теперь миддлвар, такую же функцию в файле `controllers/games.js` переименуй в `sendAllGames` 4. Удали один код, который перенесли в `middleware`, из всех контроллеров. 5. Подключи функцию `getAllGames` как `middleware` во все маршруты в файле `routes/games.js`
