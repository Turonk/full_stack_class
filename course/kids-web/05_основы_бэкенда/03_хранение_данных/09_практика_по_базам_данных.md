    ```javascript

    // Указываем, где запущена база данных
    // mongodb://localhost:27017 — адрес сервера MongoDB
    // pindie — имя базы данных на сервере MongoDB
    const DB_URL = 'mongodb://localhost:27017/pindie';
    
    async function connectToDatabase() {
      try {
    	  // Подключение к БД
        await mongoose.connect(DB_URL);
        console.log('Успешно подключились к mondodb')
      }
      catch (err) {
    	  // Обработка ошибки при подключении к базе

        console.error(err);
      }
    
      await mongoose.connect(DB_URL);
    }
    
    module.exports = connectToDatabase;
    ```

8. Теперь нужно убедиться, что настроены все миддлвары, в которые записываются функции для работы с БД. Сначала проверь категории и пользователей — с играми ситуация сложнее, и мы рассмотрим их отдельно 👇🏻

    a. В папке `middlewares` создан файл `categories.js`, куда импортирована модель `categories` и добавлена асинхронная функция `findAllCategories` для поиска всех категорий.


11. Теперь разберёмся с играми. Чтобы реализовать логику работы с ними, нужно связать данные об играх, категориях и пользователях. Проверь следующие шаги 👇🏻

    a. В схему игры добавлено поле `categories`, в котором должен храниться массив объектов, где будет указан спецтип `mongoose.Schema.Types.ObjectId` и свойство `ref` — в него записывается имя модели, на которую ссылается это поле, например `categoryModel` 

    b. В схеме содержится поле `users`. Оно должно быть аналогично полю `categories`, но в `ref` будет записана модель `userModel`

```javascript
const mongoose = require('mongoose');
// Не забываем импортировать модель, на которую ссылаемся
const userModel = require('./user');
const categoryModel = require('./category');

const gameSchema = new mongoose.Schema({
  title: {
    type: String,
    required: true,
  },
  description: {
    type: String,
    required: true,
  },
  developer: {
    type: String,
    required: true
  },
  image: {
    type: String,
    required: true
  },
  link: {
    type: String,
    required: true
  },
  // Добавляем поле для списка пользователей
  users: [{
    type: mongoose.Schema.Types.ObjectId,
    ref: userModel,
  }],
  // Добавляем поле для списка категорий
  categories: [{
    type: mongoose.Schema.Types.ObjectId,
    ref: categoryModel,
  }],
});

module.exports = mongoose.model('game', gameSchema);
```

c.  Создан файл `/middlewares/games.js`, в котором записана функция
         `findAllGames` для поиска всех игр. Чтобы получать информацию о
         категориях и пользователях, связанных с игрой, нужно использовать
         метод `populate()` 👇🏻

```javascript
// Файл middlewares/games.js

// Импортируем модель
const games = require("../models/game");

const findAllGames = async (req, res, next) => {
  // По GET-запросу на эндпоинт /games найдём все документы категорий
  // и с помощью метода populate запросим данные о связанных
  // категориях и пользователях
  req.gamesArray = await games.find({}).populate("categories").populate("users");
  next();
};

// Экспортируем функцию поиска всех игр
module.exports = findAllGames;
```

          d.  В файле `/controllers/games.js` есть функция `sendAllGames`, которая
          отправляет данные об играх 👇🏻

```javascript
// Файл controllers/games.js

const sendAllGames = (req, res) => {
  res.setHeader('Content-Type', 'application/json');
  // Вернём найденные игры в формате JSON
  res.end(JSON.stringify(req.gamesArray));
};

module.exports = sendAllGames;
```

          e.  Наконец, добавлен роут для обработки GET-запроса по адресу `/games`.
          Для этого создан отдельный файл `/routes/games.js` и в нём использованы
          раннее написанные функции `findAllGames` и `sendAllGames` 👇🏻

```javascript
// Файл routes/games.js
```
